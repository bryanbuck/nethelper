  Authentication Routes (No Auth Required)

  1. /login - User Login Page

  Purpose: User authentication interface for accessing the platform
  Component: LoginView.vue (/src/views/auth/LoginView.vue)
  API Endpoints:
  - POST /auth/login - User authentication
  - POST /auth/refresh - Token refresh

  Test Cases:
  describe('Login Page (/login)', () => {
    it('should display login form with email and password fields', () => {
      cy.visit('/login')
      cy.get('[data-cy=email-input]').should('be.visible')
      cy.get('[data-cy=password-input]').should('be.visible')
      cy.get('[data-cy=login-button]').should('be.visible')
    })

    it('should validate required fields', () => {
      cy.visit('/login')
      cy.get('[data-cy=login-button]').click()
      cy.get('.error-message').should('contain', 'Email is required')
      cy.get('.error-message').should('contain', 'Password is required')
    })

    it('should authenticate with valid credentials', () => {
      cy.visit('/login')
      cy.get('[data-cy=email-input]').type('admin@company.com')
      cy.get('[data-cy=password-input]').type('password123')
      cy.get('[data-cy=login-button]').click()
      cy.url().should('eq', '/dashboard')
    })

    it('should display error for invalid credentials', () => {
      cy.visit('/login')
      cy.get('[data-cy=email-input]').type('invalid@email.com')
      cy.get('[data-cy=password-input]').type('wrongpassword')
      cy.get('[data-cy=login-button]').click()
      cy.get('.error-alert').should('contain', 'Invalid credentials')
    })

    it('should redirect authenticated users to dashboard', () => {
      cy.login() // Custom command to set auth token
      cy.visit('/login')
      cy.url().should('eq', '/dashboard')
    })
  })

    })
